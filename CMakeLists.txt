cmake_minimum_required(VERSION 3.22)
project(maceda_eq1 VERSION 0.1.0 LANGUAGES CXX)
add_library(eq1_core src/filter.cpp)
target_include_directories(eq1_core PUBLIC src)
target_compile_features(eq1_core PUBLIC cxx_std_20)
include(CTest)
if(BUILD_TESTING)
    add_executable(eq1_tests tests/filter_test.cpp)
    target_link_libraries(eq1_tests PRIVATE eq1_core)
    add_test(NAME static_response COMMAND eq1_tests)
endif()
